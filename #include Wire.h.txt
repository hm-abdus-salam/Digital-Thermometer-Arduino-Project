#include <Wire.h>
#include <LiquidCrystal_I2C.h>

// I2C address  0x27 or 0x3F 
LiquidCrystal_I2C lcd(0x27, 16, 2);

float value;
float millivolt;
float tempC;
float tempF;

void setup()
{
  lcd.init();        // LCD initialize
  lcd.backlight();  // Backlight on
}

void loop()
{
  value = analogRead(A0);
  millivolt = (value * 5000.0) / 1024.0;
  tempC = millivolt / 10.0;
  tempF = (tempC * 9.0 / 5.0) + 32.0;

  
  lcd.setCursor(0, 0);
  lcd.print("Digital Thermometer");

  
  lcd.setCursor(0, 1);
  lcd.print("T=");
  lcd.print(tempC, 1);
  lcd.print(char(223));
  lcd.print("C ");
  lcd.print(tempF, 1);
  lcd.print(char(223));
  lcd.print("F");

  delay(2000);
}
